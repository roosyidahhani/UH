Saat ini, sistem sudah secara default menyasar ke URL deployment Anda. Jika Anda meluncurkan deploy baru dari Apps Script Anda di masa depan, silakan masukkan URL barunya di kolom input bawah ini:

Kode Google Apps Script yang Digunakan

Pastikan Apps Script Anda di Google Spreadsheet menggunakan template kode berikut ini (mendukung penulisan data dan pembacaan awan secara dua arah):

/* KODE GOOGLE APPS SCRIPT - SD NEGERI BLUBUK (2-WAY TRANSFERS) */
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Tanggal Pengiriman", "Nama Siswa", "Mata Pelajaran", "Skor", "Benar", "Salah", "Total Soal", "Detail Jawaban"]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#e2e8f0");
    }
    
    var formattedDate = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    
    sheet.appendRow([
      formattedDate,
      data.studentName,
      data.quizTitle,
      data.score,
      data.correct,
      data.wrong,
      data.totalQuestions,
      data.detail || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "status": "success", "message": "Nilai berhasil disimpan" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    var data = [];
    
    for (var i = 1; i < rows.length; i++) {
      data.push({
        timestamp: rows[i][0] ? Utilities.formatDate(new Date(rows[i][0]), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss") : "",
        studentName: rows[i][1],
        quizTitle: rows[i][2],
        score: rows[i][3],
        correct: rows[i][4],
        wrong: rows[i][5],
        totalQuestions: rows[i][6],
        detail: rows[i][7]
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({ "status": "success", "data": data }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ "status": "error", "message": error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
